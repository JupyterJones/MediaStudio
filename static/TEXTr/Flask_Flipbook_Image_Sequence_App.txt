import os
from flask import Flask, request, render_template, redirect, url_for
from werkzeug.utils import secure_filename
from PIL import Image
from moviepy.editor import ImageSequenceClip
app = Flask(__name__)
app.config['UPLOAD_FOLDER'] = 'static/projects'
app.config['MAX_CONTENT_LENGTH'] = 16 * 1024 * 1024  # 16MB limit
ALLOWED_EXTENSIONS = {'png', 'jpg', 'jpeg'}
def allowed_file(filename):
    return '.' in filename and \
           filename.rsplit('.', 1)[1].lower() in ALLOWED_EXTENSIONS
def process_images(images):
    for image in images:
        try:
            img = Image.open(image.stream)
            img = img.resize((512, 768))
            img.save(os.path.join(app.config['UPLOAD_FOLDER'], secure_filename(image.filename)))
        except (IOError, OSError) as e:
            print(f"Error processing image {image.filename}: {e}")
            #Handle the error appropriately, perhaps by displaying a message to the user.
def create_flipbook():
    image_files = [os.path.join(app.config['UPLOAD_FOLDER'], f) for f in os.listdir(app.config['UPLOAD_FOLDER']) if os.path.isfile(os.path.join(app.config['UPLOAD_FOLDER'], f)) and f.lower().endswith(('.png', '.jpg', '.jpeg'))]
    if not image_files:
        print("No images found to create flipbook.")
        return
    clip = ImageSequenceClip(image_files, durations=[2] * len(image_files))  # 2 seconds per image
    clip.write_videofile("flipbook.mp4", fps=0.5)
@app.route('/', methods=['GET', 'POST'])
def upload_file():
    if request.method == 'POST':
        if 'files[]' not in request.files:
            return redirect(request.url)
        files = request.files.getlist('files[]')
        if files:
            process_images(files)
            create_flipbook()
            return "Flipbook created successfully!"
        else:
            return "No files uploaded."
    return render_template('flipbook.html')
